## ALTER

### Add New Field

```sql
-- ADD NEW FIELD TO TABLE AT FIRST
ALTER TABLE TABLE_NAME ADD FIELD_NAME DATATYPE(N) FIRST; 
-- ADD NEW FIELD AFTER FIELD_NAME_1 FIELD
ALTER TABLE TABLE_NAME ADD FIELD_NAME DATATYPE(N) AFTER FIELD_NAME_1; 
-- ADD NEW FIELD TO TABLE AT END
ALTER TABLE TABLE_NAME ADD NEW_FIELD_NAME DATATYPE(N); 
```

### Delete Table Field

```sql
ALTER TABLE TABLE_NAME DROP FIELD_NAME; 
--DELETE LAST COLUMN
ALTER TABLE TABLE_NAME DROP LAST; 
ALTER TABLE TABLE_NAME DROP END; 
```

### Move Field

```sql
-- MOVE FIELD_NAME AFTER FIELD_NAME_2
ALTER TABLE TABLE_NAME CHANGE FIELD_NAME FIELD_NAME SAME_DATATYPE(N) AFTER FIELD_NAME_2; 
```

### Change Field Name

```sql
ALTER TABLE TABLE_NAME CHANGE FIELD_NAME FIELD_NEW_NAME SAME_DATATYPE(N);
```

### Change Field Type

```sql
ALTER TABLE TABLE_NAME CHANGE FIELD_NAME FIELD_NAME NEW_DATATYPE(N);
-- OR
ALTER TABLE TABLE_NAME MODIFY FIELD_NAME NEW_DATATYPE(N); 
```

### Change Table Name

```sql
RENAME TABLE OLD_NAME TO NEW_NAME, OLD_NAME_1 TO NEW_NAME_1;
-- OR
ALTER TABLE OLD_NAME RENAME NEW_NAME;
```

### Change Field Name Name And Type

```sql
ALTER TABLE TABLE_NAME CHANGE FIELD_NAME FIELD_NEW_NAME NEW_DATATYPE(N); 
-- OR
ALTER TABLE TABLE_NAME MODIFY NEW_FIELD_NAME NEW_DATATYPE(N); 
```

### Change Table Character Set To `utf-8`

```sql

```

### Change Table Engine

```sql
ALTER TABLE TABLE_NAME ENGINE = NEW_ENGINE_TYPE; 
```

### TRIM

```sql
TRIM([BOTH, LEADING, TRAILING] [CHAR] FROM FIELD_NAME)
-- BY DEFAULT BOTH AND THE CHAR IS SPACE
TRIM() -- REMOVE SPACES FROM LEFT AND RIGHT

TRIM(BOTH FROM FIELD_NAME)      -- REMOVE REMOVE SPACES FROM LEFT AND RIGHT
TRIM(LEADING  FROM FIELD_NAME)  -- REMOVE SPACES FROM LEFT
TRIM(TRAILING FROM FIELD_NAME)  -- REMOVE SPACES FROM RIGHT

TRIM(LEADING  '$' FROM FIELD_NAME) -- REMOVE $ FROM LEFT
TRIM(TRAILING '$' FROM FIELD_NAME) -- REMOVE $ FROM RIGHT
TRIM('$' FROM FIELD_NAME)          -- REMOVE $ FROM LEFT AND RIGHT
TRIM(BOTH '$' FROM FIELD_NAME)     -- REMOVE $ FROM LEFT AND RIGHT

RTRIM(FIELD_NAME)         -- REMOVE ONLY SPACES FROM RIGHT
LTRIM(FIELD_NAME)         -- REMOVE ONLY SPACES FROM LEFT
```

## CONTROL FLOW

### IF CONDITION

```sql
IF (CONDITION, TRUE, FALSE)
SELECT `NAME`, IF(CREATE_DATE < '2010-1-1', 'RARE', 'NOPE') AS OLDNESS FROM `USERS`;
```

### SWITCH CASE

```sql
CASE 
    WHEN CONDITION THEN RESULT
    WHEN CONDITION THEN RESULT
    WHEN CONDITION THEN RESULT
    -- .....
    ELSE RESULT
END
SELECT `NAME`,
CASE 
    WHEN CREATE_DATE < '2020-1-1' THEN 'NOPE'
    WHEN CREATE_DATE < '2010-1-1' THEN 'OLD'
    WHEN CREATE_DATE < '2000-1-1' THEN 'RARE'
    WHEN 20 THEN ': D'
    ELSE 'UNKNOWN'
END
AS OLDNESS FROM `USERS`;
```

## ORDER BY & HAVING

```sql
SELECT * FROM USERS ORDER BY TITLE;
SELECT * FROM USERS ORDER BY TITLE, POINTS;

SELECT * FROM USERS GROUP BY TITLE;
SELECT TITLE, SUM(POINTS) FROM USERS GROUP BY TITLE ORDER BY TITLE;
```

## JOIN

```sql
SELECT * FROM TABLE_NAME_1 JOIN TABLE_NAME_2;
-- THE SAME AS
SELECT * FROM TABLE_NAME_1 , TABLE_NAME_2;

-- EXAMPLE
SELECT * FROM USERS , LANGUAGES WHERE USERS.LANGUAGE_ID = LANGUAGES.LANGUAGE_ID;

-- INNER JOIN (INTERSECTION BETWEEN TWO TABLES)
SELECT * FROM USERS INNER JOIN LANGUAGES ON USERS.LANGUAGE_ID = LANGUAGES.LANGUAGE_ID;
-- SAME AS
SELECT * FROM USERS INNER JOIN LANGUAGES USING (LANGUAGE_ID);

-- LEFT JOIN (ALL ON LEFT TABLE AND INTERSECTION BETWEEN TWO TABLES)
SELECT * FROM USERS LEFT JOIN LANGUAGES ON USERS.LANGUAGE_ID = LANGUAGES.LANGUAGE_ID;
-- SAME AS
SELECT * FROM USERS LEFT JOIN LANGUAGES USING (LANGUAGE_ID);

-- RIGHT JOIN (ALL ON RIGHT TABLE AND INTERSECTION BETWEEN TWO TABLES)
SELECT * FROM USERS RIGHT JOIN LANGUAGES ON USERS.LANGUAGE_ID = LANGUAGES.LANGUAGE_ID;
-- SAME AS 
SELECT * FROM USERS RIGHT JOIN LANGUAGES USING (LANGUAGE_ID);

-- EXAMPLE
SELECT LAN.NAME, COUNT(LAN.LANGUAGE_ID) USED FROM USERS INNER JOIN LANGUAGES ON USERS.LANGUAGE_ID = LANGUAGES.LANGUAGE_ID GROUP BY LAN.LANGUAGE_ID;
```
